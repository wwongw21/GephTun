function Assert-True($Condition, [string]$Message = 'Expected true') { if (-not $Condition) { throw $Message } }
function Assert-Equal($Actual, $Expected, [string]$Message = '') {

function Assert-Throws([scriptblock]$Body) {

}

function New-Operation {
    param([string]$Action = 'Connect', [bool]$Received = $false, [bool]$Success = $false, [bool]$Exited = $false)
    $process = [pscustomobject]@{ HasExited = $Exited; ExitCode = 1; Disposed = $false }
    $process | Add-Member ScriptMethod Dispose { $this.Disposed = $true }
    $token = [guid]::NewGuid().ToString('N')
    return [pscustomobject]@{
        Action = $Action; Process = $process; Token = $token
        ResultFile = Join-Path $script:ResultDirectory ($token + '.json')
        CancelFile = Join-Path $script:ResultDirectory ($token + '.cancel.json')
        CaptureFile = Join-Path $script:TestRoot ($token + '.log')
        ResultReceived = $Received; ResultSuccess = $Success; CancelRequested = $false
        StartedUtc = [DateTime]::UtcNow; SlowNoticeShown = $false; LastResultError = ''; LastCancelError = ''
    }
}
function Wait-TestChildExit {
    # Windows PowerShell 5.1 Process.WaitForExit(int) can report false after a
    # redirected child has already exited when no wait handle was bound first.
    # Poll HasExited instead so a released fixture child is never misreported
    # as stuck (1.3.8 harness repair for the D-005 family).
    param([Parameter(Mandatory=$true)]$Process, [int]$TimeoutMilliseconds = 15000)
    $deadline = [DateTime]::UtcNow.AddMilliseconds($TimeoutMilliseconds)
    while (-not $Process.HasExited -and [DateTime]::UtcNow -lt $deadline) {
        [Threading.Thread]::Sleep(20)
    }
    return $Process.HasExited
}
function New-TestChildProcess {
    param([Parameter(Mandatory=$true)][string]$File, [string]$Arguments = '')
    $process = [Diagnostics.Process]::new()
    $hostPath = (Get-Process -Id $PID).Path
    # 1.3.8 harness repair (D-005 family): the previous Windows branch wrapped
    # the child in `cmd /c ""exe" ... -LastArg "path"""`. cmd.exe strips the
    # first and last quotes of that form, appending a stray quote to the LAST
    # argument, so the lock holder polled a file name that could never exist
    # and every native Windows run failed these cases. A .NET child process
    # with UseShellExecute=$false performs no cmd interpretation at all: %, &,
    # ^ and ! stay literal and each quoted argument round-trips verbatim.
    $childArguments = '-NoProfile -NonInteractive -File "' + $File + '"'
    if ($Arguments) { $childArguments += ' ' + $Arguments }
    $process.StartInfo.FileName = $hostPath
    $process.StartInfo.Arguments = $childArguments
    # PowerShell 7 fixtures must remain offline; Windows PowerShell 5.1 ignores these.
    $process.StartInfo.EnvironmentVariables['POWERSHELL_TELEMETRY_OPTOUT'] = '1'
    $process.StartInfo.EnvironmentVariables['DOTNET_CLI_TELEMETRY_OPTOUT'] = '1'
    $process.StartInfo.UseShellExecute = $false
    $process.StartInfo.CreateNoWindow = $true
    $process.StartInfo.RedirectStandardOutput = $true
    $process.StartInfo.RedirectStandardError = $true
    return $process
}
