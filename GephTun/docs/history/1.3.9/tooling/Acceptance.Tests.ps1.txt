function Test-Case([string]$Name,[scriptblock]$Body) {
    try { & $Body; $results.Add([pscustomobject]@{Name=$Name;Result='PASS';Detail=''}) }
    catch { $results.Add([pscustomobject]@{Name=$Name;Result='FAIL';Detail=$_.Exception.Message}) }
}
function Assert-True([bool]$Condition,[string]$Detail) { if (-not $Condition) { throw $Detail } }
function Assert-Throws([scriptblock]$Body,[string]$Expected) {
    $caught=$false
    try { $null=& $Body } catch { $caught=$true; if ($_.Exception.Message -notlike ('*'+$Expected+'*')) { throw } }
    if (-not $caught) { throw ('Expected failure: '+$Expected) }
}

function Initialize-ProviderFixture {
    $script:ProviderFixture=@{ThrowFilter='';DuplicateFilter='';Package=$null;OmitPackage=$false;FailStore='';OwnedCount=1}

function Get-FixtureFilter([string]$Name) {
    if($script:ProviderFixture.ThrowFilter -eq $Name){throw ('Provider access denied: '+$Name)}
    $value=switch($Name){
        'Address' {[pscustomobject]@{LocalAddress='Any';RemoteAddress=@('::2-fe7f:ffff:ffff:ffff:ffff:ffff:ffff:ffff')}}
        'Application' {if($script:ProviderFixture.OmitPackage){[pscustomobject]@{Program='Any'}}else{[pscustomobject]@{Program='Any';Package=$script:ProviderFixture.Package}}}
        'Service' {[pscustomobject]@{Service='Any'}}
        'Port' {[pscustomobject]@{Protocol='Any';LocalPort='Any';RemotePort='Any';IcmpType='Any';DynamicTransport='Any'}}
        'Interface' {[pscustomobject]@{InterfaceAlias='Any'}}
        'InterfaceType' {[pscustomobject]@{InterfaceType='Any'}}
        'Security' {[pscustomobject]@{Authentication='NotRequired';Encryption='NotRequired';OverrideBlockRules=$false;LocalUser='Any';RemoteUser='Any';RemoteMachine='Any'}}
    }
    $value
    if($script:ProviderFixture.DuplicateFilter -eq $Name){$value}
}

function Initialize-MetadataFixture {
    $script:MetadataFixture=@{QueryError=$false;Duplicate=$false;Missing='';BadType=''}
}
function Get-Command {
    [CmdletBinding()]param([string]$Name)
    if ($script:MetadataFixture.QueryError) { throw ('Native metadata unavailable: '+$Name) }
    $parameters=@{}
    foreach ($entry in @(
        @{Name='Name';Type=[string[]]},@{Name='Enabled';Type=[IO.FileMode]},@{Name='OverrideBlockRules';Type=[bool]},
        @{Name='ArrayMode';Type=[IO.FileMode[]]},@{Name='OptionalMode';Type=[Nullable[IO.FileMode]]}
    )) {
        if ($entry.Name -eq $script:MetadataFixture.Missing) { continue }
        $type=$entry.Type
        if ($entry.Name -eq $script:MetadataFixture.BadType) { $type='Unknown parameter type' }
        $parameters[$entry.Name]=[pscustomobject]@{ParameterType=$type}
    }
    $metadata=[pscustomobject]@{Name=$Name;CommandType='Function';ModuleName='SyntheticReadOnlyFixture';Parameters=$parameters}
    $metadata
    if ($script:MetadataFixture.Duplicate) { $metadata }
}
function New-NrptMock {
    $script:NrptMockRules=New-Object 'Collections.Generic.List[object]'
    $script:NrptMockRemoved=New-Object 'Collections.Generic.List[string]'
    $script:NrptMockAdded=0
    [pscustomobject]@{
        GetRules={ @($script:NrptMockRules.ToArray()) }
        AddRule={ param($Namespace,$NameServers,$Comment)
            $script:NrptMockAdded++
            $script:NrptMockRules.Add([pscustomobject]@{Name='fixture-'+[guid]::NewGuid().ToString('N');Namespace=@($Namespace);NameServers=@($NameServers);Comment=$Comment})
        }
        RemoveRule={ param($Name)
            $script:NrptMockRemoved.Add([string]$Name)
            $found=@($script:NrptMockRules | Where-Object {$_.Name -eq [string]$Name})
            if($found.Count -ne 1){throw ('Mock rule not found: '+$Name)}
            foreach($item in $found){[void]$script:NrptMockRules.Remove($item)}
        }
    }
}

    Test-Case 'Concurrent atomic replacement yields valid state without transient false failures' {
        $path=Join-Path $temp 'replacing.json';[IO.File]::WriteAllText($path,'{"Sequence":0}')
        $writer=[PowerShell]::Create()
        $pending=$null
        try {
            $null=$writer.AddScript({
                param($path)
                for($i=1;$i -le 1000;$i++) {
                    # Use a unique same-directory temporary file and no shared
                    # backup name. A fixed .tmp/.bak pair made this test fail
                    # because the fixture raced with its own stale backup (and
                    # antivirus), obscuring the reader behavior under test.
                    $suffix=[guid]::NewGuid().ToString('N')
                    $tmp=$path+'.'+$suffix+'.tmp'
                    $backup=$path+'.'+$suffix+'.bak'
                    $replaced=$false
                    try {
                        [IO.File]::WriteAllText($tmp,('{"Sequence":'+$i+'}'))
                        for($attempt=0;$attempt -lt 8 -and -not $replaced;$attempt++) {
                            try { [IO.File]::Replace($tmp,$path,$backup);$replaced=$true }
                            catch [IO.IOException] {
                                if($attempt -eq 7){throw}
                                Start-Sleep -Milliseconds 5
                            }
                        }
                        if(-not $replaced){throw 'Atomic replacement did not complete.'}
                    } finally {
                        if([IO.File]::Exists($tmp)){[IO.File]::Delete($tmp)}
                        if([IO.File]::Exists($backup)){[IO.File]::Delete($backup)}
                    }
                    Start-Sleep -Milliseconds 2
                }
            }).AddArgument($path)
            $pending=$writer.BeginInvoke()
            for($i=0;$i -lt 500;$i++) { $value=Read-AcceptanceJson $path;Assert-True ($null -ne $value -and $value.Sequence -ge 0 -and $value.Sequence -le 1000) 'Reader returned invalid or absent state.' }
            $null=$writer.EndInvoke($pending)
            Assert-True (-not $writer.HadErrors) ('Concurrent writer failed: '+($writer.Streams.Error -join ' | '))
            Assert-True (@(Get-ChildItem -LiteralPath (Split-Path $path) -Filter ((Split-Path $path -Leaf)+'.*.tmp') -File).Count -eq 0) 'Writer left a temporary file behind.'
            Assert-True (@(Get-ChildItem -LiteralPath (Split-Path $path) -Filter ((Split-Path $path -Leaf)+'.*.bak') -File).Count -eq 0) 'Writer left a backup file behind.'
        } finally {
            if($null -ne $pending -and -not $pending.IsCompleted){$writer.Stop();try{$null=$writer.EndInvoke($pending)}catch{} }
            $writer.Dispose()
        }
    }
