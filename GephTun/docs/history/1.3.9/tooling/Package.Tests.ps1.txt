#Requires -Version 5.1
# Isolated packaging regressions. Synthetic binaries/reports are TEST FIXTURES,
# never Windows acceptance evidence. No network, administrator actions, or real
# application files are executed. The packaging scripts themselves are exercised.
[CmdletBinding()]
param([string]$PackageDirectory, [string]$ResultJson)
Set-StrictMode -Version 2.0
$ErrorActionPreference = 'Stop'
# $PSScriptRoot is empty while param() defaults are evaluated on Windows
# PowerShell 5.1 -File runs; resolve the default in the script body instead.
if (-not $PackageDirectory) { $PackageDirectory = Split-Path -Parent $PSScriptRoot }
$script:InputRoot = [IO.Path]::GetFullPath($ExecutionContext.SessionState.Path.GetUnresolvedProviderPathFromPSPath($PackageDirectory))
# A report must bind the test harness that actually executes. Checking this before
# importing candidate helpers also makes a mismatched harness fail without running
# scripts from the selected input directory.
$candidateHarness = Join-Path $script:InputRoot 'tests/Package.Tests.ps1'
if (-not [IO.File]::Exists($candidateHarness) -or
    (Get-FileHash -LiteralPath $PSCommandPath -Algorithm SHA256).Hash -ne
    (Get-FileHash -LiteralPath $candidateHarness -Algorithm SHA256).Hash) {
    throw 'The executing package test harness differs from the source-bound harness. Invoke the harness from the selected package.'
}
. (Join-Path $script:InputRoot 'tests/Package.Common.ps1')
if ($ResultJson) {
    $script:ReportOutput = Get-GephPackageFullPath $ResultJson
    Assert-GephPackageOutput $script:ReportOutput (Get-GephPackageFullPath $script:InputRoot)
}
$script:Results = New-Object 'Collections.Generic.List[object]'
$script:RoundtripSource=$null;$script:RoundtripOutput=$null;$script:RoundtripArchive=$null;$script:RoundtripEngineering=$null
$script:RuntimeBaseline=$null;$script:RuntimeBuildArchive=$null
$script:TestRoot = Join-Path ([IO.Path]::GetTempPath()) ('GephTun-package-tests-' + [guid]::NewGuid().ToString('N'))
[void][IO.Directory]::CreateDirectory($script:TestRoot)
$script:OwnedSources = @('Verify-GephTun.ps1','tests/Package.Common.ps1','tests/build-package.ps1','tests/Validate-Source.ps1','tests/Collect-Validation.ps1','tests/Package.Tests.ps1','tests/Verify-Runtime.ps1','tests/Run-WindowsValidation.ps1')
$script:SourceSnapshot = @(foreach ($relative in $script:OwnedSources) { [pscustomobject]@{ Path = $relative; Sha256 = (Get-FileHash -LiteralPath (Join-Path $script:InputRoot $relative) -Algorithm SHA256).Hash.ToLowerInvariant() } })
$script:SuitePaths = [ordered]@{ Controller='tests/controller-results.json'; Network='tests/proxy-helper-results.json'; UI='tests/ui-results.json'; Source='tests/parser-compile-results.json'; Package='tests/package-results.json'; Acceptance='tests/acceptance-results.json' }

$script:TlsTrustCaseNames = @('TLS trusted','TLS untrusted','TLS wrong_hostname')
$script:LinkCaseNames = @('Build rejects a redirected output ancestor into the source','Package verifier rejects redirected source ancestors','Verifier rejects redirected package entries','Source validator rejects redirected result ancestors')

function Assert-Equal($Actual, $Expected, [string]$Message = '') {
    if ($Actual -ne $Expected) { throw ('{0} Expected <{1}>; got <{2}>.' -f $Message, $Expected, $Actual) }
}
function Assert-True($Actual, [string]$Message = '') { if (-not $Actual) { throw ('Assertion failed: ' + $Message) } }
function Assert-Throws([scriptblock]$Action, [string]$Pattern = '') {
    $caught = $null
    try { & $Action | Out-Null } catch { $caught = $_ }
    if ($null -eq $caught) { throw 'Expected rejection, but the operation succeeded.' }
    if ($Pattern -and $caught.Exception.Message -notmatch $Pattern) { throw ('Expected <{0}>; got <{1}>.' -f $Pattern, $caught.Exception.Message) }
}
function Write-FixtureJson([string]$Path, $Value) {
    [IO.File]::WriteAllText($Path, ($Value | ConvertTo-Json -Depth 12), (New-Object Text.UTF8Encoding($false)))
}
function Read-FixtureJson([string]$Root, [string]$Relative) { Get-Content -LiteralPath (Join-Path $Root $Relative) -Raw -Encoding UTF8 | ConvertFrom-Json }
function Get-FixtureHash([string]$Root, [string]$Relative) { (Get-FileHash -LiteralPath (Join-Path $Root $Relative) -Algorithm SHA256).Hash.ToLowerInvariant() }
function Refresh-FixtureManifest([string]$Root) {
    $entries = @(foreach ($file in @(Get-ChildItem -LiteralPath $Root -Recurse -File)) {
        $relative = $file.FullName.Substring($Root.Length + 1).Replace('\','/')
        if ($relative -eq 'PACKAGE-MANIFEST.json') { continue }
        [pscustomobject]@{ Path=$relative; Bytes=[long]$file.Length; Sha256=(Get-FileHash -LiteralPath $file.FullName -Algorithm SHA256).Hash.ToLowerInvariant() }
    })
    $release=Read-FixtureJson $Root 'RELEASE.json'
    Write-FixtureJson (Join-Path $Root 'PACKAGE-MANIFEST.json') ([pscustomobject]@{
        Package='GephTun'; Version='0.0.0'; SyntheticFixture=$true; Files=$entries
        Status=$release.Status;Date=$release.Date;InputArchiveSha256=$release.InputArchiveSha256
        Validation='tests/current-validation.json';NativeWindowsAcceptance='NOT_RUN'
    })
}

function New-Fixture {
    $root = Join-Path $script:TestRoot ([guid]::NewGuid().ToString('N'))
    Copy-Item -LiteralPath $script:Baseline -Destination $root -Recurse
    return $root
}
function Ensure-RoundtripFixture {
    # Each test that uses a built archive provisions its own prerequisite.
    # The former shared Roundtrip* variables made results depend on the
    # preceding test's success and order.
    if ($script:RoundtripSource -and $script:RoundtripArchive -and $script:RoundtripEngineering -and
        [IO.File]::Exists($script:RoundtripArchive) -and
        [IO.Directory]::Exists($script:RoundtripSource) -and
        [IO.Directory]::Exists($script:RoundtripEngineering)) { return }
    $root=New-Fixture
    $output=Join-Path $script:TestRoot ('roundtrip-prerequisite-'+[guid]::NewGuid().ToString('N'))
    $built=Invoke-Build $root $output
    $expanded=Join-Path $script:TestRoot ('roundtrip-expanded-prerequisite-'+[guid]::NewGuid().ToString('N'))
    [IO.Compression.ZipFile]::ExtractToDirectory($built.Archive,$expanded)
    $script:RoundtripSource=$root
    $script:RoundtripOutput=$output
    $script:RoundtripArchive=$built.Archive
    $script:RoundtripEngineering=Join-Path $expanded 'GephTun'
}
function Ensure-RuntimeFixture {
    Ensure-RoundtripFixture
    if ($script:RuntimeBaseline -and [IO.File]::Exists((Join-Path $script:RuntimeBaseline 'QUALIFICATION.json'))) { return }
    $output=Join-Path $script:TestRoot ('runtime-prerequisite-'+[guid]::NewGuid().ToString('N'))
    $built=& (Join-Path $script:Baseline 'tests/build-package.ps1') -SourceDirectory $script:RoundtripSource -OutputDirectory $output -RuntimeOnly
    $expanded=Join-Path $script:TestRoot ('runtime-expanded-prerequisite-'+[guid]::NewGuid().ToString('N'))
    [IO.Compression.ZipFile]::ExtractToDirectory($built.Archive,$expanded)
    $script:RuntimeBaseline=Join-Path $expanded 'GephTun'
    $script:RuntimeBuildArchive=$built.Archive
}
function Invoke-Verify([string]$Root) { & (Join-Path $script:Baseline 'Verify-GephTun.ps1') -PackageDirectory $Root -Quiet }
function Invoke-Build([string]$Root, [string]$Output) { & (Join-Path $script:Baseline 'tests/build-package.ps1') -SourceDirectory $Root -OutputDirectory $Output }
function Quote-WindowsProcessArgument([string]$Value) {
    # ProcessStartInfo.Arguments is a single command-line string on the
    # Windows PowerShell/.NET Framework host. Package paths are filesystem
    # values, never script text; NTFS path components cannot contain a quote.
    return '"' + $Value + '"'
}
function Invoke-WithRawAttributeProvider([scriptblock]$Provider,[scriptblock]$Body) {
    # PS 5.1 resolves functions dynamically. Save and restore the seam around
    # every injected filesystem provider so one attribute test cannot alter a
    # later test's command resolution.
    $saved=(Get-Command Get-GephPackageRawAttributes -CommandType Function -ErrorAction Stop).ScriptBlock
    try { Set-Item Function:\Get-GephPackageRawAttributes -Value $Provider; & $Body }
    finally { Set-Item Function:\Get-GephPackageRawAttributes -Value $saved }
}

function Test-Case([string]$Name, [scriptblock]$Body) {
    try { & $Body; $script:Results.Add([pscustomobject]@{ Result='PASS'; Test=$Name; Detail='' }) }
    catch { $script:Results.Add([pscustomobject]@{ Result='FAIL'; Test=$Name; Detail=$_.Exception.Message }) }
}
function Refresh-FixtureSourceBinding([string]$Root, [string]$Relative) {
    # This updates synthetic fixtures only. It never generates real test evidence.
    $hash=Get-FixtureHash $Root $Relative
    foreach ($reportPath in $script:SuitePaths.Values) {
        $report=Read-FixtureJson $Root $reportPath
        foreach ($source in $report.Sources) { if ($source.Path -eq $Relative) { $source.Sha256=$hash } }
        foreach ($test in $report.Tests) {
            if ((Get-GephPackageProperty $test 'File') -eq $Relative) { $test.Sha256=$hash }
        }
        Write-FixtureJson (Join-Path $Root $reportPath) $report
    }
    & (Join-Path $script:Baseline 'tests/Collect-Validation.ps1') -PackageDirectory $Root -Replace | Out-Null
    Add-FixtureQualification $Root
}
function Add-FixtureQualification([string]$Root) {
    $evidence=Read-FixtureJson $Root 'tests/current-validation.json'
    $qualification=[pscustomobject]@{
        Package='GephTun';Version='0.0.0';ProductionQualified=$false;NativeWindowsAcceptance='NOT_RUN';InputArchiveSha256=('0'*64)
        PortableRequiredSuitePassed=$evidence.RequiredTotals.Passed;PortableSupplementalPassed=$evidence.SupplementalTotals.Passed
        PortableFailed=$evidence.Totals.Failed;PortableSkipped=$evidence.Totals.Skipped;Sources=$evidence.Sources
        RequiredTotals=$evidence.RequiredTotals;SupplementalTotals=$evidence.SupplementalTotals;Totals=$evidence.Totals
        Evidence=@([pscustomobject]@{Path='tests/current-validation.json';Sha256=Get-FixtureHash $Root 'tests/current-validation.json'},
            [pscustomobject]@{Path='tests/acceptance-results.json';Sha256=Get-FixtureHash $Root 'tests/acceptance-results.json'})
    }
    Write-FixtureJson (Join-Path $Root 'QUALIFICATION.json') $qualification
    Refresh-FixtureManifest $Root
}

        try {
