#Requires -Version 5.1
[CmdletBinding()]
param([string]$PackageDirectory, [switch]$Replace, [switch]$SourceTree)
Set-StrictMode -Version 2.0
$ErrorActionPreference = 'Stop'
# $PSScriptRoot is empty while param() defaults are evaluated on Windows
# PowerShell 5.1 -File runs; resolve the default in the script body instead.
if (-not $PackageDirectory) { $PackageDirectory = Split-Path -Parent $PSScriptRoot }
. (Join-Path $PSScriptRoot 'Package.Common.ps1')
$root = Get-GephPackageFullPath $PackageDirectory
$files = Get-GephPackageInventory $root -SourceTree:$SourceTree
$release = Get-Content -LiteralPath (Join-Path $root 'RELEASE.json') -Raw -Encoding UTF8 | ConvertFrom-Json
Assert-GephPackageRelease $release
$output = Join-Path $root 'tests/current-validation.json'
Assert-GephPackageNoRedirect $output
if ((Test-Path -LiteralPath $output) -and -not $Replace) { throw 'Validation evidence already exists. Rerun the suites first, then use -Replace to update this generated evidence.' }
$mapping = [ordered]@{
    Controller = 'tests/controller-results.json'; Network = 'tests/proxy-helper-results.json'
    UI = 'tests/ui-results.json'; Source = 'tests/parser-compile-results.json'; Package = 'tests/package-results.json'
}
$supplementalMap = Get-GephPackageSupplementalMap $files
function Read-ValidationSuite([string]$Name, [string]$Relative) {
    $name = $Name; $relative = $Relative
    if (-not $files.ContainsKey($relative)) { throw ('Missing current suite report: ' + $relative) }
    $report = Get-Content -LiteralPath $files[$relative].FullName -Raw -Encoding UTF8 | ConvertFrom-Json
    $summary = Get-GephPackageSuiteSummary $report $name
    $scope = Get-GephPackageProperty $report 'Scope'
    if ([string]::IsNullOrWhiteSpace([string]$scope)) { $scope = Get-GephPackageProperty $report 'Boundary' }
    if ([string]::IsNullOrWhiteSpace([string]$scope)) { $scope = 'Portable automated checks only; native Windows acceptance not run.' }
    [pscustomobject]@{
        Name = $name; Path = $relative; Sha256 = (Get-FileHash -LiteralPath $files[$relative].FullName -Algorithm SHA256).Hash.ToLowerInvariant()
        Total = $summary.Total; Passed = $summary.Passed; Failed = $summary.Failed; Skipped = $summary.Skipped; Scope = [string]$scope
    }
}
$suites = @(foreach ($name in $mapping.Keys) { Read-ValidationSuite $name $mapping[$name] })
$supplementalSuites = @(foreach ($name in $supplementalMap.Keys) { Read-ValidationSuite $name $supplementalMap[$name] })
$sources = @(foreach ($relative in @($files.Keys | Sort-Object)) {
    if (Test-GephPackageSource $relative) { [pscustomobject]@{ Path = $relative; Sha256 = (Get-FileHash -LiteralPath $files[$relative].FullName -Algorithm SHA256).Hash.ToLowerInvariant() } }
})
$requiredTotals = Get-GephPackageTotals $suites
$supplementalTotals = Get-GephPackageTotals $supplementalSuites
$totals = Get-GephPackageTotals (@($suites) + @($supplementalSuites))
$skipped = $totals.Skipped
$evidence = [pscustomobject][ordered]@{
    Success = $true; Version = $release.Version; CapturedUtc = [DateTime]::UtcNow.ToString('o')
    Skipped = $skipped; CoverageComplete = ($skipped -eq 0)
    RequiredTotals = $requiredTotals; SupplementalTotals = $supplementalTotals; Totals = $totals
    Platform = [Environment]::OSVersion.Platform.ToString(); NativeWindowsAcceptance = 'NOT RUN'
    Suites = $suites; SupplementalSuites = $supplementalSuites; Sources = $sources
}
$null = Assert-GephPackageValidation $root $release.Version $files $evidence
$temporary = Join-Path (Join-Path $root 'tests') ('validation-' + [guid]::NewGuid().ToString('N') + '.tmp')
$backup = $temporary + '.previous'
try {
    Write-GephPackageNewJson $temporary $evidence
    # Check the snapshot once more before publishing; collection never executes tests.
    $null = Assert-GephPackageValidation $root $release.Version (Get-GephPackageInventory $root -SourceTree:$SourceTree) $evidence
    Assert-GephPackageNoRedirect $output
    if (Test-Path -LiteralPath $output) {
        if (-not $Replace) { throw 'Validation evidence appeared during collection; refusing to overwrite it.' }
        # An explicit fresh backup path works on Windows PowerShell 5.1 as well
        # as PowerShell 7; its string binder can convert a null path to ''.
        [IO.File]::Replace($temporary, $output, $backup)
    } else { [IO.File]::Move($temporary, $output) }
} finally {
    if (Test-Path -LiteralPath $temporary) { Remove-Item -LiteralPath $temporary -Force }
    if (Test-Path -LiteralPath $backup) { Remove-Item -LiteralPath $backup -Force }
}
[pscustomobject]@{ Evidence = $output; Version = $release.Version; Suites = $suites.Count; SupplementalSuites = $supplementalSuites.Count; Sources = $sources.Count; RequiredTotals = $requiredTotals; SupplementalTotals = $supplementalTotals; Totals = $totals; Skipped = $skipped; CoverageComplete = ($skipped -eq 0); NativeWindowsAcceptance = 'NOT RUN' }
