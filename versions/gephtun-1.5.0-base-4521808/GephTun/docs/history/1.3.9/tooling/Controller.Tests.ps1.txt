function Assert-Equal($Actual, $Expected, [string]$Message = '') {
    if ($Actual -ne $Expected) {
        throw "Assertion failed: $Message. Expected <$Expected>; got <$Actual>."
    }
}

function Assert-True($Actual, [string]$Message = '') {
    if (-not $Actual) { throw "Assertion failed: $Message." }
}

function Assert-Throws([scriptblock]$Action, [string]$Pattern = '') {
    $caught = $null
    try { & $Action | Out-Null } catch { $caught = $_ }
    if ($null -eq $caught) { throw 'Expected an exception, but the operation succeeded.' }
    if ($Pattern -and $caught.Exception.Message -notmatch $Pattern) {
        throw "Expected exception matching <$Pattern>; got <$($caught.Exception.Message)>."
    }
}

function New-TestModule {
    if ($null -ne $script:Module) { Remove-Module $script:Module -Force }
    $script:Module = Import-Module $ModulePath -PassThru -Force -DisableNameChecking
    [GephTun.ProxyProbe]::FailDns = $false
    [GephTun.ProxyProbe]::FailDirect = $false
    [GephTun.ProxyProbe]::DnsCalls = 0
    [GephTun.ProxyProbe]::DirectCalls = 0
    [GephTun.ProxyProbe]::ProbeCalls = 0
    [GephTun.SocksDnsRelay]::FailStart = $false
    [GephTun.SocksDnsRelay]::Last = $null
    & $script:Module {
        # Fail closed if a new code path reaches an unmocked network mutation.
        # Firewall reads are guarded too: tests must stay hermetic even though
        # reading the live firewall would not mutate anything.
        foreach ($name in @('New-NetRoute', 'Remove-NetRoute', 'Set-NetIPInterface',
            'New-NetIPAddress', 'Remove-NetIPAddress', 'Add-DnsClientNrptRule',
            'Remove-DnsClientNrptRule', 'Set-DnsClientServerAddress',
            'Disable-NetAdapterBinding', 'Enable-NetAdapterBinding',
            'New-NetFirewallRule', 'Set-NetFirewallRule', 'Remove-NetFirewallRule',
            'Set-NetFirewallProfile', 'Get-NetFirewallRule', 'Get-NetFirewallProfile',
            'Get-NetFirewallAddressFilter', 'Get-NetFirewallApplicationFilter', 'Get-NetFirewallServiceFilter',
            'Get-NetFirewallPortFilter', 'Get-NetFirewallInterfaceFilter', 'Get-NetFirewallInterfaceTypeFilter',
            'Get-NetFirewallSecurityFilter',
            'Stop-Process', 'Start-Process', 'Clear-DnsClientCache', 'Set-DnsClient',
            'Register-ScheduledTask', 'Unregister-ScheduledTask', 'Get-ScheduledTask',
            'New-ScheduledTaskAction', 'New-ScheduledTaskTrigger', 'New-ScheduledTaskPrincipal', 'New-ScheduledTaskSettingsSet')) {
            Set-Item -Path ('Function:script:' + $name) -Value {
                [CmdletBinding()]
                param([Parameter(ValueFromRemainingArguments = $true)]$Arguments)
                throw 'Test safety guard: an unmocked Windows mutation was attempted.'
            }
        }
    }
}

function Test-Case([string]$Name, [scriptblock]$Body) {
    try {
        New-TestModule
        & $Body
        $script:Results.Add([pscustomobject]@{ Result = 'PASS'; Test = $Name; Detail = '' })
    } catch {
        $result = 'FAIL'
        if ($_.Exception.Data['GephTunFixtureSkip'] -eq $true) { $result = 'SKIP' }
        $script:Results.Add([pscustomobject]@{ Result = $result; Test = $Name; Detail = $_.Exception.Message })
    }
}

function New-ControllerTestLink([string]$Path, [string]$Target) {
    try { New-Item -Path $Path -ItemType SymbolicLink -Target $Target -ErrorAction Stop | Out-Null }
    catch {
        # Windows PowerShell 5.1 can reject fixture creation before a product
        # function is reached. Only the explicit privilege error is skippable;
        # an arbitrary access denial or failed product assertion remains FAIL.
        $cause = $_.Exception.GetBaseException()
        $privilegeRequired = ($cause -is [ComponentModel.Win32Exception] -and $cause.NativeErrorCode -eq 1314) -or
            (($cause.HResult -band 65535) -eq 1314) -or
            ($_.FullyQualifiedErrorId -match '^NewItemSymbolicLinkElevationRequired(?:,|$)')
        if ([Environment]::OSVersion.Platform -eq [PlatformID]::Win32NT -and $privilegeRequired) {
            $skip = New-Object InvalidOperationException('BLOCKED_BY_HOST: Windows denied symbolic-link fixture creation because the required privilege is absent. Rerun this case from the approved elevated test host; its redirection protection is NOT_RUN here.')
            $skip.Data['GephTunFixtureSkip'] = $true
            throw $skip
        }
        throw
    }
}

function New-BinaryFixture {
    $folder = Join-Path $script:TestRoot ([guid]::NewGuid().ToString('N'))
    [IO.Directory]::CreateDirectory((Join-Path $folder 'bin')) | Out-Null
    foreach ($name in @('tun2socks-windows-amd64.exe', 'wintun.dll')) {
        $bytes = New-Object byte[] 256
        $bytes[0] = 0x4d; $bytes[1] = 0x5a
        $bytes[0x3c] = 0x80
        $bytes[0x80] = 0x50; $bytes[0x81] = 0x45
        $bytes[0x84] = 0x64; $bytes[0x85] = 0x86
        [IO.File]::WriteAllBytes((Join-Path (Join-Path $folder 'bin') $name), $bytes)
    }
    $entries = @(foreach ($name in @('tun2socks-windows-amd64.exe', 'wintun.dll')) {
        [pscustomobject]@{
            File = $name
            Sha256 = (Get-FileHash -LiteralPath (Join-Path (Join-Path $folder 'bin') $name) -Algorithm SHA256).Hash
        }
    })
    $manifest = [pscustomobject]@{ Binaries = $entries }
    [IO.File]::WriteAllText((Join-Path $folder 'DEPENDENCIES.json'), (ConvertTo-Json $manifest -Depth 6))
    # 1.3.8 (D-009): the boot recovery task action must resolve inside the
    # redirected package root during tests.
    [IO.File]::WriteAllText((Join-Path $folder 'GephTun-BootReconcile.ps1'), '# test stub')
    & $script:Module {
        param($Folder)
        $script:PackageRoot = $Folder
        function script:Get-AuthenticodeSignature {
            [CmdletBinding()]
            param([string]$LiteralPath)
            [pscustomobject]@{ Status = 'Valid' }
        }
    } $folder
    return [pscustomobject]@{ Folder = $folder; Manifest = $manifest }
}

function Initialize-ContainmentFixture {
    $folder = Initialize-RecoveryFixture
    & $script:Module {
        $script:FirewallRules = @()

        $script:RecoveryDeny = ''; $script:RecoveryDenyUsed = $false
        function script:Invoke-RecoveryDenial([string]$Stage) {
            if ($Stage -eq $script:RecoveryDeny -and -not $script:RecoveryDenyUsed) {
                $script:RecoveryDenyUsed = $true
                throw ('Injected recovery permission denial at ' + $Stage)
            }
        }
        function script:Get-NetFirewallProfile {
            [CmdletBinding()] param($PolicyStore)
            $script:FirewallProfiles
        }
        function script:Set-NetFirewallProfile {
            # NetSecurity's Enabled parameter is an enum, not bool. A typed
            # double exposes bool-to-enum conversion failures on restoration.
            [CmdletBinding()] param($Name, [GephTun.ControllerGpoBoolean]$Enabled, $PolicyStore)
            if ([string]$Enabled -eq 'False') { Invoke-RecoveryDenial 'profile' }
            $script:Events.Add([pscustomobject]@{ Operation = 'SetProfile'; Value = ($Name + '=' + $Enabled) })
            foreach ($profile in $script:FirewallProfiles) { if ($profile.Name -eq $Name) { $profile.Enabled = [string]$Enabled } }
        }
        function script:Get-NetFirewallRule {
            [CmdletBinding()] param($Name, $PolicyStore)
            $script:LastFirewallPolicyStore = $PolicyStore
            if ($PolicyStore -eq 'ActiveStore' -and $script:InactiveRule) { return @() }
            $matches = @($script:FirewallRules | Where-Object { -not $Name -or $_.Name -eq $Name })
            if ($Name -and $matches.Count -eq 0) { Write-Error 'No MSFT_NetFirewallRule objects found.' }
            $matches
        }
        function script:New-NetFirewallRule {
            [CmdletBinding()] param($Name, $DisplayName, $Description, $Direction, $Action, $Profile, $RemoteAddress, $Enabled, $PolicyStore, [ValidateSet('NotRequired','Required','NoEncap')]$Authentication, [ValidateSet('NotRequired','Required','Dynamic')]$Encryption, [bool]$OverrideBlockRules)
            $script:LastFirewallPolicyStore = $PolicyStore
            $rule = [pscustomobject]@{ Name = $Name; DisplayName = $DisplayName; Direction = $Direction; Action = $Action; Profile = $Profile; Enabled = [string]$Enabled; RemoteAddress = @($RemoteAddress) }
            $script:FirewallRules = @($script:FirewallRules) + @($rule)
            $script:Events.Add([pscustomobject]@{ Operation = 'NewFirewall'; Value = $rule })
            $rule
        }
        function script:Set-NetFirewallRule {
            [CmdletBinding()] param($Name, $RemoteAddress, $PolicyStore)
            foreach ($rule in $script:FirewallRules) { if ($rule.Name -eq $Name) { $rule.RemoteAddress = @($RemoteAddress) } }
            $script:Events.Add([pscustomobject]@{ Operation = 'SetFirewall'; Value = $Name })
        }
        function script:Remove-NetFirewallRule {
            [CmdletBinding()] param($Name, $PolicyStore)
            $script:FirewallRules = @($script:FirewallRules | Where-Object { $_.Name -ne $Name })
            $script:Events.Add([pscustomobject]@{ Operation = 'RemoveFirewall'; Value = $Name })
        }
        function script:Get-NetFirewallAddressFilter {
            [CmdletBinding()] param([Parameter(ValueFromPipeline=$true)]$InputObject)
            process { [pscustomobject]@{ LocalAddress = @('Any'); RemoteAddress = @($InputObject.RemoteAddress) } }
        }
        function script:Get-NetFirewallApplicationFilter {

            process { [pscustomobject]@{ InterfaceAlias = 'Any' } }
        }
        function script:Get-NetFirewallInterfaceTypeFilter {
            [CmdletBinding()] param([Parameter(ValueFromPipeline=$true)]$InputObject)
            process { [pscustomobject]@{ InterfaceType = 'Any' } }
        }
        function script:Get-NetFirewallSecurityFilter {
            [CmdletBinding()] param([Parameter(ValueFromPipeline=$true)]$InputObject)
            process {
                if ($script:SecurityReadMode -eq 'denied') { throw 'Injected SecurityFilter permission denial.' }
                if ($script:SecurityReadMode -eq 'missing') { return }
                $security = $script:SecurityPersistent
                if ($script:LastFirewallPolicyStore -eq 'ActiveStore') { $security = $script:SecurityActive }
                $security
                if ($script:SecurityReadMode -eq 'duplicate') { $security }
            }
        }
    }
    return $folder
}

    Test-Case '500 atomic status replacements remain complete while the controller reads concurrently' {
        $folder = Initialize-RecoveryFixture
        $path = Join-Path $folder 'heartbeat.json'
        & $script:Module { param($Path) Write-GephTunJson $Path @{ Revision = 0; End = 0; Payload = ('snapshot-0:' * 64) } } $path
        & $script:Module {
            # Exercise real metadata enumeration during replacement while the
            # ACL boundary remains a hermetic provider fixture on every host.
            function script:Get-Acl {
                [CmdletBinding()] param($LiteralPath)
                $acl = [pscustomobject]@{}
                $acl | Add-Member ScriptMethod GetOwner { param($Type) [pscustomobject]@{ Value = 'S-1-5-32-544' } }
                $acl | Add-Member ScriptMethod GetAccessRules { param($Explicit,$Inherited,$Type) [pscustomobject]@{ IdentityReference = [pscustomobject]@{ Value = 'S-1-5-32-544' }; AccessControlType = 'Allow'; FileSystemRights = [Security.AccessControl.FileSystemRights]::FullControl } }
                $acl
            }
        }
        $state = [hashtable]::Synchronized(@{ Written = 0 })
        $ready = New-Object Threading.ManualResetEventSlim($false)
        $start = New-Object Threading.ManualResetEventSlim($false)
        $writer = [powershell]::Create()
        $job = $null
        try {
            $writerScript = {
                param($ModulePath, $Path, $State, $Ready, $Start)
                $ErrorActionPreference = 'Stop'
                Import-Module $ModulePath -Force -DisableNameChecking
                $Ready.Set()
                if (-not $Start.Wait(5000)) { throw 'Concurrent reader did not start.' }
                for ($revision = 1; $revision -le 500; $revision++) {
                    Write-GephTunJson $Path @{ Revision = $revision; End = $revision; Payload = (('snapshot-' + $revision + ':') * 64) }
                    $State.Written = $revision
                }
            }
            $writer.AddScript($writerScript.ToString()).AddArgument($ModulePath).AddArgument($path).AddArgument($state).AddArgument($ready).AddArgument($start) | Out-Null
            $job = $writer.BeginInvoke()
            Assert-True ($ready.Wait(5000)) 'writer reached the concurrent barrier'
            $start.Set()
            $deadline = [DateTime]::UtcNow.AddSeconds(30)
            do {
                $overlap = -not $job.IsCompleted
                $snapshot = & $script:Module { param($Path) Read-GephTunJson $Path } $path
                Assert-Equal $snapshot.Revision $snapshot.End 'one complete revision'
                Assert-Equal $snapshot.Payload (('snapshot-' + $snapshot.Revision + ':') * 64) 'complete payload from that same revision'
                $script:JsonStress.ReadSnapshots++
                if ($overlap) { $script:JsonStress.ConcurrentReadSnapshots++ }
                if ($script:JsonStress.ReadSnapshots % 5 -eq 1) {
                    $scanOverlap = -not $job.IsCompleted
                    & $script:Module { param($Folder) Assert-GephTunStorageTreePermissions $Folder } $folder
                    $script:JsonStress.StorageScans++
                    if ($scanOverlap) { $script:JsonStress.ConcurrentStorageScans++ }
                }
                if ([DateTime]::UtcNow -ge $deadline) { throw 'Concurrent snapshot test exceeded its time bound.' }
            } while (-not $job.IsCompleted -or $script:JsonStress.ReadSnapshots -lt 500)
            $writer.EndInvoke($job) | Out-Null
            Assert-Equal $writer.Streams.Error.Count 0
            $script:JsonStress.CompletedWrites = $state.Written
            Assert-Equal $state.Written 500
            Assert-True ($script:JsonStress.ConcurrentReadSnapshots -gt 0)
            Assert-True ($script:JsonStress.ConcurrentStorageScans -gt 0) 'at least one protected-state scan overlaps real writes'
            $last = & $script:Module { param($Path) Read-GephTunJson $Path } $path
            Assert-Equal $last.Revision 500
            Assert-Equal @(Get-ChildItem -LiteralPath $folder -Filter '*.tmp').Count 0
        } finally {
            if ($null -ne $job -and -not $job.IsCompleted) { $writer.Stop() }
            $writer.Dispose(); $ready.Dispose(); $start.Dispose()
        }
    }

    Test-Case '40 repeated connections retry denied DNS route and profile recovery from the saved journal' {
        # Reuse one state directory so stale disconnect requests and already
        # restored profile intent are exercised across successive connections.
        $folder = Initialize-StartupFixture
        for ($cycle = 0; $cycle -lt 40; $cycle++) {
            $mode = @('normal', 'dns', 'route', 'profile')[$cycle % 4]
            $disabled = $cycle % 3
            & $script:Module {
                param($Mode, $Disabled)
                $script:Events.Clear(); $script:SavedSessions.Clear()
                $script:RecoveryDeny = $Mode; $script:RecoveryDenyUsed = $false
                foreach ($profile in $script:FirewallProfiles) { $profile.Enabled = 'True' }
                $script:FirewallProfiles[$Disabled].Enabled = 'False'
            } $mode $disabled
            $connected = & $script:Module { Start-GephTunSession 9909 }
            Assert-Equal $connected.Success $true "recovery cycle $cycle connects over old request"
            $oldToken = & $script:Module { $script:Session.Token }
            if ($mode -ne 'normal') {
                Assert-Throws { & $script:Module { Restore-GephTunSession } } 'recovery permission denial'
                $pending = ConvertFrom-Json ([IO.File]::ReadAllText((Join-Path $folder 'session.json')))
                Assert-Equal $pending.Token $oldToken "cycle $cycle preserves journal ownership"
                if ($mode -in @('dns', 'route')) {
                    Assert-Equal (& $script:Module { $script:EngineAlive }) $true "cycle $cycle retains engine while diversion cleanup is denied"
                    Assert-Equal ([GephTun.SocksDnsRelay]::Last.Disposed) $false "cycle $cycle retains DNS relay"
                    Assert-Equal (& $script:Module { $script:FirewallRules.Count }) 1 "cycle $cycle retains containment"
                    Assert-Equal @($pending.Routes | Where-Object { $_.Kind -eq 'Bypass' }).Count 1 "cycle $cycle retains the upstream bypass"
                } else {
                    Assert-Equal @($pending.Routes).Count 0 'profile denial happens after diversion is gone'
                    Assert-Equal @($pending.Containment.Profiles).Count 1 'only the unrestored profile remains pending'
                    Assert-Equal (& $script:Module { $script:EngineAlive }) $false
                }
                # Model retry by a later recovery action reading the persisted
                # partial journal, while preserving actual remaining surfaces.
                & $script:Module { $script:Session = $null }
            }
            & $script:Module { Restore-GephTunSession }
            $state = & $script:Module { [pscustomobject]@{ Routes = $script:NetworkRoutes.Count; Rules = $script:FirewallRules.Count; Dns = $script:DnsRules.Count; Addresses = $script:AdapterAddresses.Count; Engine = $script:EngineAlive; Session = $script:Session; Profiles = @($script:FirewallProfiles | ForEach-Object { $_.Enabled }) } }
            foreach ($surface in @('Routes', 'Rules', 'Dns', 'Addresses')) { Assert-Equal $state.$surface 0 "recovery cycle $cycle $surface" }
            Assert-Equal $state.Engine $false
            Assert-Equal $state.Session $null
            Assert-Equal $state.Profiles[$disabled] 'False'
            Assert-Equal @($state.Profiles | Where-Object { $_ -eq 'False' }).Count 1
            Assert-Equal (Test-Path -LiteralPath (Join-Path $folder 'session.json')) $false
            Assert-Equal ([GephTun.SocksDnsRelay]::Last.Disposed) $true
            & $script:Module { param($Token) Write-GephTunJson (Join-Path $script:FixtureRoot 'disconnect.json') @{ Token = $Token } } $oldToken
            $script:RecoveryStressCycles.Add([pscustomobject]@{ Cycle = $cycle + 1; Scenario = $mode; Result = 'PASS' })
        }
    }

    Test-Case '100 complete user connection cycles leave no journal or owned network state after success failure and cancellation' {
        $stages = @('route:203.0.113.7/32', 'engine-created', 'binding', 'ip-interface', 'ip-address', 'adapter-dns', 'dns-registration', 'route:0.0.0.0/1', 'route:128.0.0.0/1', 'nrpt', 'dns-probe')
        $cycleClock = [Diagnostics.Stopwatch]::StartNew()
        for ($cycle = 0; $cycle -lt 100; $cycle++) {
            # Reset fakes between independent users, while running the real
            # startup, containment, journaling and recovery functions each time.
            $folder = Initialize-StartupFixture
            $mode = @('success', 'fault', 'cancel')[$cycle % 3]
            $stage = $stages[$cycle % $stages.Count]
            & $script:Module {
                param($Mode, $Stage, $Cycle)
                if ($Mode -eq 'fault') { $script:FaultAt = $Stage }
                if ($Mode -eq 'cancel') { $script:CancelAt = $Stage }
                # Rotate the originally disabled profile so recovery proves
                # exact profile restoration under every cycle outcome.
                $script:FirewallProfiles[$Cycle % 3].Enabled = 'False'
            } $mode $stage $cycle
            if ($mode -eq 'success') {
                $connected = & $script:Module { Start-GephTunSession 9909 }
                Assert-Equal $connected.Success $true "cycle $cycle connects"
                & $script:Module { Restore-GephTunSession }
            } elseif ($mode -eq 'fault') {
                Assert-Throws { & $script:Module { Start-GephTunSession 9909 } } 'Injected startup failure'
            } else {
                Assert-Throws { & $script:Module { Start-GephTunSession 9909 } } 'cancelled'
            }
            $state = & $script:Module { [pscustomobject]@{ Routes = $script:NetworkRoutes.Count; Rules = $script:FirewallRules.Count; Dns = $script:DnsRules.Count; Addresses = $script:AdapterAddresses.Count; Engine = $script:EngineAlive; Session = $script:Session; Profiles = @($script:FirewallProfiles | ForEach-Object { $_.Enabled }) } }
            foreach ($surface in @('Routes', 'Rules', 'Dns', 'Addresses')) { Assert-Equal $state.$surface 0 "cycle $cycle $surface" }
            Assert-Equal $state.Engine $false "cycle $cycle engine"
            Assert-Equal $state.Session $null "cycle $cycle session"
            Assert-Equal $state.Profiles[$cycle % 3] 'False' "cycle $cycle restored profile"
            Assert-Equal @($state.Profiles | Where-Object { $_ -eq 'False' }).Count 1 "cycle $cycle only the original profile is disabled"
            Assert-Equal (Test-Path -LiteralPath (Join-Path $folder 'session.json')) $false "cycle $cycle journal"
            Assert-Equal ([GephTun.SocksDnsRelay]::Last.Disposed) $true "cycle $cycle relay"
            $script:StressCycles.Add([pscustomobject]@{ Cycle = $cycle + 1; Scenario = $mode; Stage = $(if ($mode -eq 'success') { 'complete' } else { $stage }); Result = 'PASS' })
        }
